Interpreter - .tflite 모델을 실제로 실행해주는 도구
(ex) interpreter = Interpreter(model_path = modelPath)

interpreter.allocate_tensors() - 모델이 사용할 입출력 tensor의 메모리를 할당 (= 모델실행에 필요한 공간 준비) , 이걸 실행후 set_tensor() / get_tensor() 사용가능
* set_tensor() - 모델에게 입력 데이터 전달
(ex) interpreter.set_tensor(입력_tensor의_인덱스, 입력데이터)
* get_tensor() - 모델이 계산한 결과를 가저오는것
(ex) interpreter.get_tensor(출력_tensor의_인덱스)

interpreter.get_input_details() - 모델의 입력 tensor 정보
 -> 안에 tensor의 인덱스, 데이터 타입, shape.... 등 
interpreter.get_output_details() - 모델의 출력 tensor 정보
 -> 출력 tensor정보
=> 이거 하는 이유 : 모델에 무엇을 넣을지, 무엇이 나오는지 확인하는것

input_dtype = input_details[0]['dtype'] # 위에서 받은 입력_디테일에서 tensor가 어떤 데이터 타입을 요구하는지 확인
height = input_details[0]['shape'][1] # (이미지라면) 이미지의 높이 가져오기
width = input_details[0]['shape'][2] # 이미지의 너비 가져오기

ansToText = {0:'scissors', 1:'rock', 2:'paper'} # 출력번호 -> 내가 원하는 이름
colorList = [(255,0,0),(0,255,0),(0,0,255)] -> 색 / (B,G,R)
IMG_SIZE = 224

def processImage(frame):
cv2.cvtColor(이미지, 변환방법) - 색상을 변환하는 함수
    img = cv2.cvtColor(frame,cv2.COLOR_BGR2RGB) # BGR을 RGB로 변경

    # 모델의 입력 형태로 수정: (1,224,224,3)
    # Normalization 처리 안함 (모델 내에서 -1 ~ 1로 변환 처리함)
cv2.resize(이미지,(가로 크기, 세로 크기)) - 이미지 크기 변경    
    img = cv2.resize(img, (IMG_SIZE, IMG_SIZE)) 

np.enpand_dimas(배열,axis=추가할 위치) : 배열에 차원을 한개 추가
추가 이유: 모델이 batch(한번에 처리할 묶음 속의 갯수)를 요구해서.
    img = np.expand_dims(img, 0) # 이미지 1장의 shape (224,224,3)에 batch 차원을 추가 → (1,224,224,3)

    # 모델에 입력하여 결과 얻기
    #   input tensor 설정
img.astype() - 이미지를 모델이 요구하는 데이터 타입으로 변경
interpreter.set_tensor() - 모델의 입력에 이미지 넣기
    interpreter.set_tensor(input_details[0]['index'], img.astype(input_dtype))

interpreter.invoke() - 모델 실행
    interpreter.invoke()

    #   output tensor 얻기
    output_data = interpreter.get_tensor(output_details[0]['index'])[0] # 여기서 잘 모르겟으면 프린트로 찍어보면서 쓰기

np.argmax() - 가장 큰값 위치 반환
    ans = np.argmax(output_data)

    text = ansToText[ans] # 0,1,2 중 한개 받으면 그거에 맞는 ansToText = {0:'scissors', 1:'rock', 2:'paper'} 반환

    # 판정 결과 표시
cv2.putText(이미지, 글자, 위치, 폰트, 크기, 색상, 두께) - 이미지 위에 글자쓰기
    cv2.putText(frame,text,(180,50),cv2.FONT_HERSHEY_PLAIN,2,colorList[ans],2)

# 카메라 설정
cap = cv2.VideoCapture(0) # 0번 카메라 열기
cap.set(cv2.CAP_PROP_FRAME_WIDTH,320) # 영상 너비 320
cap.set(cv2.CAP_PROP_FRAME_HEIGHT,240) # 높이 240
cap.set(cv2.CAP_PROP_BUFFERSIZE,1) # 프레임버퍼 크기 1

# 윈도우 설정
cv2.namedWindow('cam', cv2.WINDOW_NORMAL) # cam 이라는 이름의 창 만들기 , cv2.WINDOW_NORMAL : 창크기 조절 가능 옵션
cv2.resizeWindow('cam', 320+40, 240+60) # 창크기 360*300

startTime = time.time() # 현재시간 저장
while(cap.isOpened()): # 카메라가 열려있는동안 반복해라
    ret,frame=cap.read() # 한장의 프레임에서 ret,frame 가져와
    if not ret: break # ret : 이미지를 제대로 가져왓는가? TRUE OR FALSE

    # 이미지 처리
    processImage(frame) # 위에서 만든 함수 지금 가져온 frame 넣어서 실행

    # FPS 표시
    curTime = time.time() # 현재 시간
    fps = 1/(curTime - startTime) # 이전 프레임부터 현재 프레임까지 걸린시간을 이용해 fps계산
    startTime = curTime
    cv2.putText(frame,f'FPS: {fps:.1f}',(20, 50),cv2.FONT_HERSHEY_PLAIN,2,(0,255,255),2) # 화면에 프레임 표시

    # 이미지 출력
    cv2.imshow('cam',frame) # 현재 frame 에는 이미 ai 결과(가위바위보), fps가 적혀있음

	 # 10ms 동안 키 입력 대기
    key = cv2.waitKey(10)
    if  key == ord('q'): break

cap.release() # 카메라 닫기
cv2.destroyAllWindows() # 모든 창 닫기
